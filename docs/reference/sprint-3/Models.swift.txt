import Foundation
import FirebaseFirestore

// MARK: - Companies (companies/{companyId})
struct Company: Codable, Identifiable {
    @DocumentID var id: String?
    var name: String
    var logoURL: String
    var joinCode: String
    var emailDomain: String
    var city: String
    var officeLatitude: Double
    var officeLongitude: Double
    var createdAt: Date
}
